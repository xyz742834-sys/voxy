 static Map<String,Object> trace()throws Exception{
  McNativeIngestTrace.setEnabled(true);McNativeIngestTrace.observeLeaf(OTHER);McNativeIngestTrace.observeLeaf(LEAF);
  var storage=new Storage();var e=new WorldEngine(storage);
  e.setSaveCallback((into,s,nonBlocking,acquired)->{s.setNotDirty();if(acquired)s.release();return true;});
  var pin=e.acquire(0,1,3,2);WorldUpdater.insertUpdate(e,section(2,true));
  McNativeIngestTrace.observeLeaf(LEAF);long before=McNativeIngestTrace.now(),bg=McNativeIngestTrace.generation();
  McNativeIngestTrace.ingestQueued(2,7,4,e);McNativeIngestTrace.dirtyConsumed(LEAF,e);McNativeIngestTrace.meshStarted(LEAF);McNativeIngestTrace.meshed(LEAF,e);
  var old=snapshot(e,before,bg,GLASS,GLASS);old.put("target_ingest_executed",false);check(old.get("update")==null,"enqueue only cannot change");
  McNativeIngestTrace.observeLeaf(LEAF);before=McNativeIngestTrace.now();bg=McNativeIngestTrace.generation();McNativeIngestTrace.ingestQueued(2,7,4,e);
  // Actual updater changes a DIFFERENT 16-block chunk section in the SAME 32-block leaf.
  WorldUpdater.insertUpdate(e,section(3,true));McNativeIngestTrace.dirtyConsumed(LEAF,e);McNativeIngestTrace.meshStarted(LEAF);
  long meshRead=pin._unsafeGetRawDataArray()[TARGET];check(meshRead==GLASS,"read still glass");
  var field=WorldUpdater.class.getDeclaredField("observer");field.setAccessible(true);var observer=(WorldUpdater.Observer)field.get(null);
  var written=new CountDownLatch(1);var release=new CountDownLatch(1);var failure=new AtomicReference<Throwable>();
  // Pause only between actual target content write and its observer/dirty publication. No invented outcome.
  WorldUpdater.setObserver((into,leaf,changed)->{if(Thread.currentThread().getName().equals("review-target-update")){written.countDown();try{release.await();}catch(InterruptedException x){throw new RuntimeException(x);}}observer.leafUpdated(into,leaf,changed);});
  var t=new Thread(()->{try{WorldUpdater.insertUpdate(e,section(2,false));}catch(Throwable x){failure.set(x);}},"review-target-update");
  Map<String,Object> race;
  try{
   t.start();check(written.await(2,TimeUnit.SECONDS),"target reached observer");check(pin._unsafeGetRawDataArray()[TARGET]==AIR,"real updater wrote air");
   McNativeIngestTrace.meshed(LEAF,e);race=snapshot(e,before,bg,GLASS,pin._unsafeGetRawDataArray()[TARGET]);
   race.put("meshTargetRaw",meshRead);race.put("target_updater_published",false);race.put("update_origin_chunk_x",3);race.put("target_chunk_x",2);
  }finally{release.countDown();t.join(2000);WorldUpdater.setObserver(observer);}
  check(!t.isAlive()&&failure.get()==null,"target child cleanup "+failure.get());race.put("owned_child_joined",true);
  pin.release();e.free();McNativeIngestTrace.setEnabled(false);
  return Map.of("old_enqueue_only",old,"unrelated_change_delayed_target_publication",race);
 }
 static Map<String,Object> shutdown()throws Exception{
  var storage=new Storage();var v=new VoxyInstance(){protected SectionStorage createStorage(WorldIdentifier id){return storage;}};
  var e=new WorldEngine(storage,v);e.markActive();var f=VoxyInstance.class.getDeclaredField("activeWorlds");f.setAccessible(true);((Map)f.get(v)).put(null,e);

  var t=new Thread(()->{try{v.shutdown();}catch(Throwable x){error.set(x);}endFlag.set(Thread.currentThread().isInterrupted());},"review-owned-instance-shutdown");
